/* ESP32 Wi-Fi Hotspot to SSC-32 Servo Controller Bridge (6 Servos) Hardware Connections: ESP32 TX2 (GPIO 17) -> SSC-32 RX (through logic level shifter if needed) ESP32 GND -> SSC-32 GND */ #include #include const char* ssid = "SSC32_Robot_Hotspot"; const char* password = "12345678"; WebServer server(80); void handleCommand() { // Enable CORS headers so browser fetch works without restrictions server.sendHeader("Access-Control-Allow-Origin", "*"); if (server.hasArg("text")) { String cmd = server.arg("text"); // Forward command to SSC-32 via Serial2 (RX=16, TX=17) Serial2.print(cmd); server.send(200, "text/plain", "OK: " + cmd); } else { server.send(400, "text/plain", "Missing text argument"); } } void handleStatus() { server.sendHeader("Access-Control-Allow-Origin", "*"); server.send(200, "text/plain", "ESP32 SSC-32 Bridge Active"); } void setup() { Serial.begin(115200); // Initialize UART2 for SSC-32 at 115200 baud (change if your SSC-32 baud is 9600/38400) Serial2.begin(115200, SERIAL_8N1, 16, 17); WiFi.softAP(ssid, password); IPAddress myIP = WiFi.softAPIP(); Serial.print("AP IP address: "); Serial.println(myIP); server.on("/command", handleCommand); server.on("/status", handleStatus); server.begin(); Serial.println("HTTP server started"); } void loop() { server.handleClient(); }
