ESP32 Arduino Firmware Code

Flash this Arduino code to your ESP32 using the ESP32Servo library and WebSocketsServer library (by Markus Sattler). 

#include <WiFi.h>
#include <WebSocketsServer.h>
#include <ESP32Servo.h>

// Wi-Fi Access Point Credentials
const char* ssid = "ESP32-Servo-Controller";
const char* password = "password123";

WebSocketsServer webSocket = WebSocketsServer(81);

Servo servoX; // Connected to X-axis
Servo servoY; // Connected to Y-axis

const int PIN_SERVO_X = 18; // GPIO 18
const int PIN_SERVO_Y = 19; // GPIO 19

void webSocketEvent(uint8_t num, WStype_t type, uint8_t * payload, size_t length) {
    if(type == WStype_TEXT) {
        // Expecting JSON string: {"x": 12.5, "y": -4.2}
        String message = "";
        for(int i = 0; i < length; i++) {
            message += (char)payload[i];
        }

        // Simple JSON Parsing for x and y
        int xIdx = message.indexOf("\"x\":");
        int yIdx = message.indexOf("\"y\":");

        if(xIdx != -1 && yIdx != -1) {
            float angleX = message.substring(xIdx + 4).toFloat();
            float angleY = message.substring(yIdx + 4).toFloat();

            // Map angles (-90 to +90 degrees) to servo pulse width (0 to 180 degrees)
            int servoValX = map((int)constrain(angleX, -90, -90), -90, 90, 0, 180);
            int servoValY = map((int)constrain(angleY, -90, 90), -90, 90, 0, 180);

            servoX.write(servoValX);
            servoY.write(servoValY);
        }
    }
}

void setup() {
    Serial.begin(115200);

    // Setup AP Mode
    WiFi.softAP(ssid, password);
    Serial.println("AP IP address: ");
    Serial.println(WiFi.softAPIP());

    // Attach Servos
    ESP32PWM::allocateTimer(0);
    ESP32PWM::allocateTimer(1);
    servoX.setPeriodHertz(50);
    servoY.setPeriodHertz(50);
    servoX.attach(PIN_SERVO_X, 500, 2400);
    servoY.attach(PIN_SERVO_Y, 500, 2400);

    // Start WebSocket server
    webSocket.begin();
    webSocket.onEvent(webSocketEvent);
    Serial.println("WebSocket server started on port 81");
}

void loop() {
    webSocket.loop();
}
